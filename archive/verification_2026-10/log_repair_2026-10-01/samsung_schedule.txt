 function showActiveSlideText() {
 // 모든 텍스트 숨기기
 document.querySelectorAll(".slide_text .item").forEach(text => {
 text.style.display = "none";
 });

 // 현재 활성화된 슬라이드 찾기
 let activeSlide = document.querySelector(".swiper-slide-active");

 if (activeSlide) {
 let textId = activeSlide.getAttribute("data-id");
 let targetText = document.getElementById(textId);

 if (targetText) {
 targetText.style.display = "block"; // 해당 텍스트 보이기
 }
 }
 }

 // 정지 버튼
 document.querySelector(".stop").addEventListener("click", function () {
 document.querySelector(".stop").style.display = "none";
 document.querySelector(".play").style.display = "block";
 swiper.autoplay.stop();
 });

 // 재생 버튼
 document.querySelector(".play").addEventListener("click", function () {
 document.querySelector(".stop").style.display = "block";
 document.querySelector(".play").style.display = "none";
 swiper.autoplay.start();
 });

 // 초기 텍스트 표시
 showActiveSlideText();
 });
