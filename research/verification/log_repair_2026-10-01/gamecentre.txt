 function setGameEtc() {
 
 var game = $("ul.game-list-n > li.on");
 //game.attr("game_sc", "2");
 //game.attr("sr_id", "9");

 var gameDate = game.attr("g_dt");
 var leId = game.attr("le_id");
 var srId = game.attr("sr_id");
 var season = game.attr("season");
 var gameSc = game.attr("game_sc");
 var entry = game.attr("entry_ck");
 var startPitcher = game.attr("start_ck");
 var vodCheck = game.attr("vod_ck");
 var gameScId = game.attr("game_sc_id");
 
 if (game.html() == undefined) { // 전체 취소
 $("#gameCenterContents").children().remove();

 //$("#gameComment").text("당일 경기는 모두 취소되었습니다.");
 //$("#gameComment").show();
 } else {
 if (leId == "1") {

 // 하이라이트 영상 체크
 if ($("object#player").length > 0 && getBrowserInfoStr() == "Explorer") {
 //alert(getBrowserInfoStr())
 this.player = document.getElementById("player");
 this.player.url = "";
 this.player.controls.pause();
 }

 // 정규시즌 또는 국제전의 경우 srId 변경
 if (srId.indexOf("0") > -1 || srId.indexOf("8") > -1 || srId.indexOf("9") > -1) {
 srId = "0,8,9";
 }

 if (srId == "1" && gameSc == "1") { // 시범경기는 경기전 프리뷰 미표출
 $("#tabDepth2").children().remove();
 $("#gameCenterContents").children().remove();

 $("#gameComment").show();

 } else if ((srId == "0,8,9" && game.attr("sr_id") == "9" && (gameSc == "1" || gameSc == "2")) || (game.attr("sr_id") == "8" && gameSc == "2" && gameScId == "41")) { // 올스타전, 평가전
 $("#tabDepth2").children().remove();
 $("#gameCenterContents").children().remove();

 var comment = "";

 if (gameSc == "1") { // 프리뷰
 comment = "올스타전은 프리뷰가 제공되지 않습니다.";
 } else if (gameSc == "2") { // 경기중
 comment = "경기 종료 후 오늘 경기 기록을 확인하실 수 있습니다.";
 }

 $("#gameComment").text(comment);
 $("#gameComment").show();
 } else if (srId == "0,8,9" && game.attr("sr_id") == "8" && ((gameSc == "1" || gameSc == "2" || game.attr("g_id").indexOf("KR") == -1))) { // 국제경기
 $("#tabDepth2").children().remove();
 $("#gameCenterContents").children().remove();
 
 var comment = "";

 if (gameSc == "1") {
 comment = "국제대회는 프리뷰를 제공하지 않습니다.";
 } else if (gameSc == "2") {
 comment = "국제대회는 키플레이어를 제공하지 않습니다.";
 } else if (game.attr("g_id").indexOf("KR") == -1) {
 comment = "리뷰는 한국 경기만 제공합니다.";
 }

 $("#gameComment").text(comment);
 $("#gameComment").show();
 } else if (srId == "0,8,9" && gameSc == "1" && new Date() -1 && gameScId != "41") { // 21.07.15 | veron | 국제대회 리뷰 미표출
 // $("#tabDepth2").children().remove();
 // $("#gameCenterContents").children().remove();
 // $("#gameComment").text("국제대회는 리뷰를 제공하지 않습니다.");
 // $("#gameComment").show();

 //} else {
 } else {
 $("#gameComment").hide();
 
 srId = game.attr("sr_id"); // 2017-07-14 yeeun, 올스타전 예외처리 위해서

 setGameTab(leId, srId, season, gameSc, entry, startPitcher, vodCheck);

 // 탭 파라미터 체크
 if (section != "") {
 $("#tabDepth2 > li").siblings().removeClass("on");
 $("#tabDepth2 > li[section=" + Request("section") + "]").addClass("on");
 setGameDetailSection(Request("section"));
 section = "";

 history.pushState(null, null, "/Schedule/GameCenter/Main.aspx"); // 기존 section 파라미터가 다른 경기 이동시 영향을 끼침
 } else {
 setGameDetail();
 }
 }
 } else {
 $("#gameCenterContents").children().remove();

 $("#gameComment").hide();
 }
 }
 }

 // 경기(프리뷰/ 키플레이어 / 리뷰 / 하이라이트 등) 탭 선택
 function setGameTab(leId, srId, season, gameSc, entry, startPitcher, vodCheck) {
 var ul = $("#tabDepth2");
 ul.removeClass();
 ul.addClass("tab");
 ul.children().remove();

 if (gameSc == "1") { // 경기전 : 프리뷰
 ul.append($(" 프리뷰 "));
 } else if (gameSc == "2" || gameSc == "5") { // 경기중, 서스펜디드 : 키플레이어
 ul.append($(" 키플레이어 "));
 } else { // 경기종료 : 키플레이어 / 리뷰 / 하이라이트
 if (srId != "8" && srId != "9" && parseInt(season) >= 2016) {
 ul.append($(" 키플레이어 "));
 }
 ul.append($(" 리뷰 "));

 if (vodCheck == 1 && season >= 2016)
 {
 ul.append($(" 하이라이트 "));
 }
 }

 ul.addClass(String.format("tab{0}", ul.children("li").length));
 ul.children("li").eq(0).addClass("on");
 }

 // 브라우저확인
 function getBrowserInfoStr() {
 var strBrowser = '';
 var ua = navigator.userAgent;
 if (ua.indexOf("Firefox") != -1) strBrowser = "Firefox";
 else if (ua.indexOf("Chrome") != -1) strBrowser = "Chrome";
 else if (ua.indexOf("MSIE") != -1) strBrowser = "Explorer";
 else strBrowser = "ETC";
 return strBrowser;
 }

 // 프리뷰 하단 TAB
 function setPreviewTab(section, awayPit, homePit) {
 var ul = $("#tabPreview");
 ul.children().remove();

 switch(section)
 {
 case "START_PIT": // 선발투수 전력분석
 if (awayPit == undefined){
 ul.append($(" 팀 전력비교 "));
 }
 else{
 ul.append($(" 선발투수 전력분석 "));
 ul.append($(" 팀 전력비교 "));
 }

 ul.append($(" 라인업 분석 "));
 break;
 case "TEAM": // 팀 전력비교
 if (awayPit == undefined) {
 ul.append($(" 팀 전력비교 "));
 }
 else {
 ul.append($(" 선발투수 전력분석 "));
 ul.append($(" 팀 전력비교 "));
 }
 ul.append($(" 라인업 분석 "));
 break;
 case "LINEUP": // 라인업 분석
 if (awayPit == undefined) {
 ul.append($(" 팀 전력비교 "));
 } else {
 ul.append($(" 선발투수 전력분석 "));
 ul.append($(" 팀 전력비교 "));
 }
 ul.append($(" 라인업 분석 "));
 break;
 }

 ul.show()
 }

 // 선수 검색
 function getPlayerSearch() {
 if ($("#txtTopPlayerWord").val() == "") {
 alert("선수명은 필수 입니다.");
 $("#txtTopPlayerWord").focus();
 return false;
 } else {
 //var searchWord = encodeURI($("#txtTopPlayerWord").val());

 //if (searchWord != "") {
 // location.href = "/Player/Search.aspx?searchWord=" + searchWord;
 //}
 var request = $.ajax({
 type: "post"
 , url: "/ws/Controls.asmx/GetSearchPlayer"
 , dataType: "json"
 , data: {
 name: $("#txtTopPlayerWord").val()
 }
 , error: ajaxFailed
 , async: true
 });

 request.done(function (data) {
 if (data.code == "100") {
 var container = $(".search-result");
 container.children().remove();

 container.hide();

 if (data.now.length > 0 || data.retire.length > 0) {
 // 2017-05-22 yeeun, 홍지희 사원 요청, 결과가 1명일 경우 바로 페이지 연결
 if (data.now.length + data.retire.length == 1) {
 var link = "";

 if (data.now.length == 1) {
 link = data.now[0]["P_LINK"];
 } else {
 link = data.retire[0]["P_LINK"];
 }

 location.href = link;
 } else {
 if (data.now.length > 0) {
 container.append($(" 등록 선수 "));

 var div = $(" ");
 div.children().remove();

 setPlayerDiv(data.now, div, true);

 container.append(div);
 }

 if (data.retire.length > 0) {
 container.append($(" 역대 선수 "));

 var div = $(" ");
 div.children().remove();

 setPlayerDiv(data.retire, div, false);

 container.append(div);
 }

 //container.append($(String.format(" '{0}' 검색결과 모두 보기 ", $("#txtTopPlayerWord").val())));
 container.append($(" '" + $("#txtTopPlayerWord").val() + "' 검색결과 모두 보기 "));
 container.show();
 }
 } else {
 container.append($(" 검색 결과가 없습니다. "));
 container.show();
 }
 }
 });
 }
 }

 // 선수 검색 div 
 function setPlayerDiv(list, div, isActive) {
 $.each(list, function (index, player) {
 var aLink = $(" ");
 aLink.attr("href", player["P_LINK"]);

 var dl = $(" ");
 var dt = $(" ");
 var backNo = "";

 if (isActive) {
 //backNo = String.format("No.{0}", player["BACK_NO"]);
 backNo = "No." + player["BACK_NO"];
 }

 //dt.text(String.format("{0} {1}", player["P_NM"], backNo));
 dt.text(player["P_NM"] + " " + backNo);
 dl.append(dt);

 var dd = $(" ");
 //dd.text(String.format("{0} {1}({2})", player["T_NM"], player["POS_NO"], player["P_TYPE"]));
 dd.text(player["T_NM"] + " " + player["POS_NO"] + "(" + player["P_TYPE"] + ")");
 dl.append(dd);

 var ddImg = $(String.format(" ", "//6ptotvmi5753.edge.naverncp.com/KBO_IMAGE/KBOHome"));
 dl.append(ddImg);

 aLink.append(dl);
 div.append(aLink);
 });
 }
